+//! Test drivers that tamper with `Hdr` and `Ct2` chunks in transit, for the runtime monitor
+//! (feature `monitor`).
+//!
+//! Honest runs never deliver a bad MAC, so a monitor of honest traces cannot see a MAC check
+//! whose result the code ignores. These drivers deliver tampered chunks and log each one with a
+//! `tamper` event (`crate::monitor::tamper`). A `recv` that fails leaves the party in its old
+//! state (`GenericMessagingScka::messaging_scka_recv` keeps the old state on an error), and the
+//! run goes on.
+//!
+//! - `tamper_chaos` tampers with chunks that complete a decoder, so that the honest code fails
+//!   at the MAC check of that `recv`, and then delivers the original chunk.
+//! - `tamper_early` tampers with a chunk that does not complete the decoder. The honest code
+//!   accepts it and keeps it in the decoder, and every later decode of that epoch fails at the
+//!   MAC check: the party stays in that epoch for the rest of the session.
+//!
+//! Whether a chunk completes the decoder comes from a dry run on a clone of the receiver's
+//! state with the honest chunk and no events written. The dry run does not depend on the MAC
+//! check of a tampered chunk.

+fn copy_msg(m: &Message) -> Message {

+        MessagePayload::Hdr(c) => MessagePayload::Hdr(*c),
+        MessagePayload::Ek(c) => MessagePayload::Ek(*c),
+        MessagePayload::EkCt1Ack(c) => MessagePayload::EkCt1Ack(*c),
+        MessagePayload::Ct1Ack(b) => MessagePayload::Ct1Ack(*b),
+        MessagePayload::Ct1(c) => MessagePayload::Ct1(*c),
+        MessagePayload::Ct2(c) => MessagePayload::Ct2(*c),
+    };

+    }

+/// The kind of a `Hdr` or `Ct2` message.
+fn kind(m: &Message) -> Option<Kind> {

+    }

+/// A copy of the `Hdr` or `Ct2` message `m` with one random byte of the chunk changed.
+fn tampered<R: CryptoRng>(m: &Message, rng: &mut R) -> Message {
+    let mut t = copy_msg(m);
+    let c: &mut Chunk = match &mut t.payload {
+        MessagePayload::Hdr(c) | MessagePayload::Ct2(c) => c,
+        _ => unreachable!("only Hdr and Ct2 chunks are tampered with"),
+    };
+    let i = rng.next_u32() as usize % c.data.len();
+    c.data[i] ^= rng.random_range(1..=255u8);
+    t
+}

+/// Whether `state` adds the chunk of `m` to its decoder: a `Hdr` chunk in `NoHeaderReceived`

/-! # Checking one session

For each party of a session the monitor:

1. splits the party's events into operations (`op`): operation 0 is the initialization, every
   other operation ends with exactly one step event;
2. builds one table per operation from that operation's primitive calls, and adds the decoder
   result and parity chunk of the operation, read at the model state before it (the learning
   pass, not trusted);
3. runs the verified checker `BraidModel.checkRun` from `initAliceO` / `initBobO` over the
   pairs (table, observation). Its result is the verdict; `BraidModel.accepted_alice` and
   `BraidModel.accepted_bob` state what an acceptance means.

It also checks that each table entry is a function across the party's operations, and, for the
session, the delivery of messages, the agreement of keys and the reuse of randomness (not
covered by the theorems). A `tamper` event (a test changed a message in transit) is not a
primitive call: the party check skips it, and the delivery check reads it. -/

/-- One operation of one party: its number, its events, and its step event (none for the
initialization). -/
structure Operation where
  op : Nat
  events : Array Event
  step : Option Event
  deriving Inhabited

/-- The result of checking one party. -/
structure PartyResult where
  party : String
  operations : Nat
  verdict : Except String Unit
  /-- Secrets by epoch from the accepted observations. -/
  keys : List Key

/-- Group a party's events into operations and check their shape. -/
def operations (evs : Array Event) : Except String (Array Operation) := do
  let mut ops : Array Operation := #[]
  let mut cur : Option Operation := none
  for e in evs do
    match cur with
    | some o =>
      if o.op == e.op then
        if o.step.isSome then
          throw s!"line {e.line}: event after the step event of operation {o.op}"
        let step := if e.isStep then some e else none
        cur := some ⟨o.op, o.events.push e, step⟩
      else
        if e.op < o.op then throw s!"line {e.line}: operation {e.op} after operation {o.op}"
        if o.op != 0 && o.step.isNone then
          throw s!"operation {o.op} has no step event"
        ops := ops.push o
        cur := some ⟨e.op, #[e], if e.isStep then some e else none⟩
    | none =>
      if e.op != 0 then throw s!"line {e.line}: the first operation is {e.op}, not 0"
      cur := some ⟨e.op, #[e], if e.isStep then some e else none⟩
  match cur with
  | some o =>
    if o.op != 0 && o.step.isNone then throw s!"operation {o.op} has no step event"
    ops := ops.push o
  | none => pure ()
  return ops

/-- Insert into a table; a different value for a known key is an error. -/
def insertFn {α β : Type} [BEq α] [Hashable α] [BEq β] (what : String) (m : HashMap α β)
    (k : α) (v : β) : Except String (HashMap α β) :=
  match m[k]? with
  | some v' =>
    if v' == v then .ok m else .error s!"{what} returned two different values for one argument"
  | none => .ok (m.insert k v)

/-- Add one primitive call to a table. Events that are not primitive calls are skipped. -/
def addCall (T : Tables) (e : Event) : Except String Tables := do
  let a := e.args
  match e.f with
  | "hkdf" =>
    let k := (← jBytes (← arg a 0), ← jBytes (← arg a 1), ← jBytes (← arg a 2), ← jNat (← arg a 3))
    let m ← insertFn "hkdf" T.hkdf k (← jBytes e.ret)
    return { T with hkdf := m }
  | "hmac" =>
    let k := (← jBytes (← arg a 0), ← jBytes (← arg a 1))
    let m ← insertFn "hmac" T.hmac k (← jBytes e.ret)
    return { T with hmac := m }
  | "mlkem.generate" =>
    let r ← match e.ret with | .arr r => pure r | _ => throw "generate: bad result"
    let v := (← jBytes (← arg r 0), ← jBytes (← arg r 1), ← jBytes (← arg r 2))
    let m ← insertFn "generate" T.keyGen (← jBytes (← arg a 0)) v
    return { T with keyGen := m }
  | "mlkem.encaps1" =>
    let r ← match e.ret with | .arr r => pure r | _ => throw "encaps1: bad result"
    let v := (← jBytes (← arg r 0), ← jBytes (← arg r 1), ← jBytes (← arg r 2))
    let k := (← jBytes (← arg a 0), ← jBytes (← arg a 1))
    let m ← insertFn "encaps1" T.encaps1 k v
    return { T with encaps1 := m }
  | "mlkem.encaps2" =>
    -- logged as `[ek, es]`, keyed as `(es, ek)` like the model's `Encaps2`
    let k := (← jBytes (← arg a 1), ← jBytes (← arg a 0))
    let m ← insertFn "encaps2" T.encaps2 k (← jBytes e.ret)
    return { T with encaps2 := m }
  | "mlkem.decaps" =>
    let k := (← jBytes (← arg a 0), ← jBytes (← arg a 1), ← jBytes (← arg a 2))
    let m ← insertFn "decaps" T.decaps k (← jBytes e.ret)
    return { T with decaps := m }
  | "mlkem.ek_matches_header" =>
    -- logged as `[ek, hdr]`, keyed as `(hdr, ek)` like the model's `ekMatches`
    let k := (← jBytes (← arg a 1), ← jBytes (← arg a 0))
    let m ← insertFn "ek_matches_header" T.ekMatches k (← e.ret.getBool?)
    return { T with ekMatches := m }
  | _ => return T

/-- Add a decoder result to a table. -/
def addMessage (T : Tables) (d : MDec) (m : Option Bytes) : Except String Tables := do
  let t ← insertFn "decoder" T.message d m
  return { T with message := t }

/-- Add a parity chunk to a table. -/
def addParity (T : Tables) (enc : MEnc) (c : MChunk) : Except String Tables := do
  let t ← insertFn "encoder" T.parity enc c
  return { T with parity := t }

/-- The decoder that `receiveEq` adds the chunk of `m` to in state `σ`, with the chunk added. -/
def decoderAfter (σ : State MEnc MDec) (m : Msg MChunk) : Option MDec :=
  if m.epoch != σ.epoch then none else
  let add (d : MDec) (c : MChunk) : MDec := (d.1, d.2 ++ [c])
  match σ, m.payload with
  | .headerSent _ _ _ d _, .ct1 c => some (add d c)
  | .ekSentCt1Received _ _ _ _ d, .ct2 c => some (add d c)
  | .noHeaderReceived _ _ d, .hdr c => some (add d c)
  | .ct1Sampled _ _ _ _ _ _ d, .ek c => some (add d c)
  | .ct1Sampled _ _ _ _ _ _ d, .ekCt1Ack c => some (add d c)
  | .ct1Acknowledged _ _ _ _ _ d, .ek c => some (add d c)
  | .ct1Acknowledged _ _ _ _ _ d, .ekCt1Ack c => some (add d c)
  | _, _ => none

/-- The encoder that `send` takes the next chunk from in state `σ`. Fresh encoders (transitions
(1) and (7)) start with a systematic chunk and need no entry. -/
def encoderOf : State MEnc MDec → Option MEnc
  | .keysSampled _ _ _ _ _ enc | .headerSent _ _ _ _ enc | .ct1Received _ _ _ _ enc
  | .ct1Sampled _ _ _ _ _ enc _ | .ekReceivedCt1Sampled _ _ _ _ _ _ enc
  | .ct2Sampled _ _ enc => some enc
  | _ => none

/-- The message of an observation. -/
def obsMsg : Obs MChunk → Msg MChunk
  | .send _ m .. | .recvOk m .. | .recvErr m => m

/-- A short description of a state. -/

/-- A short description of a message. -/

    | .none => "none" | .ct1Ack => "ct1ack"
    | .hdr c => s!"hdr #{c.index}" | .ek c => s!"ek #{c.index}"
    | .ekCt1Ack c => s!"ekct1ack #{c.index}" | .ct1 c => s!"ct1 #{c.index}"

    | .keysUnsampled e a =>
      match Q.keyGen r with

      | some kg => match Auth.macHdrO Q a e kg.2.1 with

        | some _ => "no chunk of the header encoder"
    | .headerReceived e a hdr _ =>

          | none => "no hkdf call for KDF_AUTH"

    | _ => "no parity chunk for the encoder"

    | none => "a primitive value of a payload without a decoder (ct2 encoder)"

        if k' != k && k'.map (·.1) == k.map (·.1) then some "key bytes" else none,
        if σ'.label != l then some s!"state (model {repr σ'.label}, code {repr l})" else none,
        if σ'.epoch != e then some s!"epoch (model {σ'.epoch}, code {e})" else none]

/-- The randomness of an operation: the seed of its `generate` call or the randomness of its

/-- Check one party of a session. -/
def checkParty (party : String) (evs : Array Event) : PartyResult := Id.run do
  let fail (msg : String) (n : Nat) : PartyResult := ⟨party, n, .error msg, []⟩
  let ops ← match operations evs with
    | .ok ops => pure ops
    | .error err => return fail err 0
  let some init := ops[0]? | return fail "no initialization" 0
  if init.op != 0 then return fail "no initialization (operation 0)" 0
  if let some e := init.events.find? (·.isStep) then
    return fail s!"line {e.line}: step event in the initialization (operation 0)" 0
  let some ie := init.events.find? (·.f == "init") | return fail "no init event" 0
  if (init.events.filter (·.f == "init")).size != 1 then
    return fail "two init events in the initialization" 0
  let role ← match (arg ie.args 0).bind Json.getStr? with
    | .ok r => pure r
    | .error err => return fail err 0
  let expected := if party == "A" then "a" else "b"
  if role != expected then
    return fail s!"line {ie.line}: party {party} initializes with role {role}, not {expected}" 0
  let ss ← match (arg ie.args 1).bind jBytes with
    | .ok s => pure s
    | .error err => return fail err 0
  -- `all` collects every entry of the party, to check that each primitive is a function.
  let mut all : Tables := {}
  let mut T0 : Tables := {}
  for e in init.events do
    match addCall T0 e, addCall all e with
    | .ok t, .ok a => T0 := t; all := a
    | .error err, _ | _, .error err => return fail err 0
  let Q0 := T0.prims
  let σ0? := if party == "A" then initAliceO Q0 ss else initBobO Q0 ss
  let some σ0 := σ0? | return fail "initialization: no hkdf call for the authenticator" 0
  -- learning pass: one table per operation, replayed with the checker's own step function
  let mut steps : Array (OPrims MEnc MDec MChunk × Obs MChunk) := #[]
  let mut σ? : Option (State MEnc MDec) := some σ0
  let mut firstFail : Option String := none
  for o in ops[1:] do
    let some se := o.step | return fail s!"operation {o.op} has no step event" steps.size
    let r ← match randomness o with
      | .ok r => pure r
      | .error err => return fail err steps.size
    let obs ← match parseObs se r with
      | .ok obs => pure obs
      | .error err => return fail s!"line {se.line}: {err}" steps.size
    let mut T : Tables := {}
    for e in o.events do
      match addCall T e, addCall all e with
      | .ok t, .ok a => T := t; all := a
      | .error err, _ | _, .error err => return fail s!"line {e.line}: {err}" steps.size
    if let some σ := σ? then
      -- the state before the call, as the code logged it, is the model's state
      let pre : Except String (Label × Nat) := do
        return (← parseLabel (← (← arg se.args 0).getStr?), ← jNat (← arg se.args 1))
      match pre with
      | .error err => return fail s!"line {se.line}: {err}" steps.size
      | .ok (l, ep) =>
        if l != σ.label || ep != σ.epoch then
          return fail s!"operation {o.op} (line {se.line}): the code's state before the call is \
            {repr l} epoch {ep}, the model's is {State.describe σ}" steps.size
      -- the decoder result of a receive, at the decoder the model adds the chunk to
      match obs, decoded o with
      | .recvOk m .. , .ok (some (pts, res)) | .recvErr m, .ok (some (pts, res)) =>
        if let some d := decoderAfter σ m then
          if d.1 != 2 * pts then
            return fail s!"operation {o.op}: decoder of {d.1} bytes, code needs {pts} points"
              steps.size
          match addMessage T d res, addMessage all d res with
          | .ok t, .ok a => T := t; all := a
          | .error err, _ | _, .error err => return fail s!"operation {o.op}: {err}" steps.size
      | _, .error err => return fail err steps.size
      | _, _ => pure ()
      -- the parity chunk of a send
      match obs, encoderOf σ with
      | .send _ m .., some enc =>
        if enc.2 ≥ enc.1.length / 32 then
          let c? : Option MChunk := match m.payload with
            | .hdr c | .ek c | .ekCt1Ack c | .ct1 c | .ct2 c => some c
            | _ => none
          if let some c := c? then
            if c.index != enc.2 % 65536 || c.data.length != 32 then
              return fail s!"operation {o.op}: parity chunk #{c.index} of {c.data.length} bytes, \
                expected index {enc.2 % 65536} and 32 bytes" steps.size
            match addParity T enc c, addParity all enc c with
            | .ok t, .ok a => T := t; all := a
            | .error err, _ | _, .error err => return fail s!"operation {o.op}: {err}" steps.size
      | _, _ => pure ()
      let Q := T.prims
      match checkStep Q σ obs with
      | some σ' => σ? := some σ'
      | none =>
        let why := explainStep Q σ obs
        let other := if (why.splitOn "hmac").length > 1 && T.hmac.size > 0 then
            s!" (this operation logged {T.hmac.size} hmac call(s), on other keys or data)"
          else ""
        firstFail := some s!"operation {o.op} (line {se.line}), {se.f} in {State.describe σ}, \
          message {describeMsg (obsMsg obs)}: {why}{other}"
        σ? := none
    steps := steps.push (T.prims, obs)
  -- the verdict of the verified checker
  match checkRun σ0 steps.toList with
  | some _ =>
    if let some f := firstFail then
      return fail s!"internal: the learning pass failed but checkRun accepts: {f}" steps.size
    ⟨party, steps.size, .ok (), obsKeys (steps.toList.map (·.2))⟩
  | none =>
    fail (firstFail.getD "rejected (no diagnosis)") steps.size

/-- The result of the session-level checks. -/
structure SessionChecks where
  delivery : Except String Unit
  keys : Except String Unit
  incompleteEpochs : List Nat
  randomness : Except String Unit

/-- Whether `t` is a tampered copy of `m`: a chunk message with the same epoch, payload kind,
chunk index and chunk length, and other chunk bytes. -/
def tamperedCopy (m t : Msg MChunk) : Bool :=
  let chunk : Payload MChunk → Option MChunk
    | .hdr c | .ek c | .ekCt1Ack c | .ct1 c | .ct2 c => some c
    | _ => none
  match chunk m.payload, chunk t.payload with
  | some c, some d =>
    m.epoch == t.epoch && m.payload.ctorIdx == t.payload.ctorIdx && c.index == d.index &&
      c.data.length == d.data.length && c.data != d.data
  | _, _ => false

/-- Delivery, key agreement and reuse of randomness for a session's events, in file order.

Delivery: each receive consumes an equal earlier send of the other party. A `tamper` event of a
party, with arguments `[orig, copy]`, says that the network made `copy` from `orig` and that the
party receives `copy` in the same operation. `orig` must be an earlier send of the other party
that no receive consumed (a send, not a tampered copy), and `copy` a tampered copy of it
(`tamperedCopy`). The operation must have one tamper event, and its receive must be of `copy`.
That receive consumes no send: `orig` stays available. -/
def sessionChecks (evs : Array Event) (ka kb : List Key) : SessionChecks := Id.run do
  -- delivery: each receive consumes an equal earlier send of the other party, or the copy of the
  -- tamper event of its own operation
  let mut sent : HashMap (String × String) Nat := {}
  -- the tamper event waiting for its receive: line, party, operation, copy
  let mut pending : Option (Nat × String × Nat × String) := none
  let mut delivery : Except String Unit := .ok ()
  let mut seen : HashSet String := {}
  let mut randomness : Except String Unit := .ok ()
  for e in evs do
    if let some (l, p, op, _) := pending then
      if delivery.isOk && (e.party != p || e.op != op) then
        delivery := .error s!"line {l}: the operation of the tamper event does not receive the \
          tampered copy"
        pending := none
    if e.f == "States::send" then
      if let .ok m := e.ret.getObjVal? "msg" then
        let k := (e.party, m.compress)
        sent := sent.insert k (sent.getD k 0 + 1)
    else if e.f == "States::recv" && delivery.isOk then
      if let some m := e.args[2]? then
        let peer := if e.party == "A" then "B" else "A"
        match pending with
        | some (l, _, _, c) =>
          if m.compress != c then
            delivery := .error s!"line {e.line}: party {e.party} received another message than \
              the tampered copy of the tamper event on line {l}"
          pending := none
        | none =>
          let k := (peer, m.compress)
          match sent[k]? with
          | some (n + 1) => sent := sent.insert k n
          | _ => delivery := .error s!"line {e.line}: party {e.party} received a message that \
              party {peer} did not send"
    else if e.f == "tamper" && delivery.isOk then
      let peer := if e.party == "A" then "B" else "A"
      match e.args[0]?, e.args[1]? with
      | some o, some t =>
        match parseMsg o, parseMsg t with
        | .ok mo, .ok mt =>
          if pending.isSome then
            delivery := .error s!"line {e.line}: two tamper events in one operation"
          else if sent.getD (peer, o.compress) 0 == 0 then
            delivery := .error s!"line {e.line}: party {e.party} received a tampered copy of a \
              message that party {peer} did not send"
          else if !tamperedCopy mo mt then
            delivery := .error s!"line {e.line}: the tampered message differs from the original \
              in more than the chunk bytes"
          else
            pending := some (e.line, e.party, e.op, t.compress)
        | _, _ => delivery := .error s!"line {e.line}: bad message in a tamper event"
      | _, _ => delivery := .error s!"line {e.line}: a tamper event needs two messages"
    else if (e.f == "mlkem.generate" || e.f == "mlkem.encaps1") && randomness.isOk then
      let i := if e.f == "mlkem.generate" then 0 else 1
      if let some r := e.args[i]? then
        let s := r.compress
        if seen.contains s then
          randomness := .error s!"line {e.line}: {e.f} reuses randomness of an earlier call"
        seen := seen.insert s
  if let some (l, _, _, _) := pending then
    if delivery.isOk then
      delivery := .error s!"line {l}: the operation of the tamper event does not receive the \
        tampered copy"
  -- keys: at most one per epoch and party, equal where both parties have one
  let toMap (ks : List Key) : Except String (HashMap Nat Bytes) :=
    ks.foldlM (init := {}) fun m (ep, s) =>
      if m.contains ep then .error s!"two keys for epoch {ep}" else .ok (m.insert ep s)
  let keys : Except String Unit × List Nat := match toMap ka, toMap kb with
    | .error err, _ => (.error s!"party A: {err}", [])
    | _, .error err => (.error s!"party B: {err}", [])
    | .ok ma, .ok mb =>
      let eps := (ma.keys ++ mb.keys).eraseDups.mergeSort
      eps.foldl (init := (.ok (), [])) fun (r, inc) ep =>
        match ma[ep]?, mb[ep]? with
        | some a, some b => if a == b || !r.isOk then (r, inc)
          else (.error s!"epoch {ep}: the parties' secrets differ", inc)
        | _, _ => (r, inc ++ [ep])
  ⟨delivery, keys.1, keys.2, randomness⟩

end Monitor

/-! # Reading the event log

The Rust feature `monitor` (`src/monitor.rs`) writes one JSON object per line:
`{"s":session,"p":party,"op":operation,"f":function,"args":[..],"ret":..}`, with bytes as
lowercase hex. This module parses the lines into `Event`s and the step events into
observations (`BraidModel.Obs`). -/

open Lean Aeneas Aeneas.Std BraidModel

namespace Monitor

/-- One logged call. `line` is its line number in the file, from 1. -/
structure Event where
  line : Nat
  session : Nat
  party : String
  op : Nat
  f : String
  args : Array Json
  ret : Json
  deriving Inhabited

/-- The value of a hex digit. -/
def hexDigit (c : Char) : Option Nat :=
  if '0' ≤ c ∧ c ≤ '9' then some (c.toNat - '0'.toNat)
  else if 'a' ≤ c ∧ c ≤ 'f' then some (c.toNat - 'a'.toNat + 10)
  else none

/-- The bytes of a lowercase hex string. -/
def decodeHex (s : String) : Except String Bytes := do
  let rec go : List Char → Array Std.U8 → Except String (Array Std.U8)
    | [], acc => .ok acc
    | a :: b :: rest, acc =>
      match hexDigit a, hexDigit b with
      | some x, some y => go rest (acc.push (byte (16 * x + y)))
      | _, _ => .error s!"bad hex digit in {s.take 16}"
    | [_], _ => .error "odd number of hex digits"
  return (← go s.toList #[]).toList

/-- The bytes of a JSON hex string. -/
def jBytes (j : Json) : Except String Bytes := do decodeHex (← j.getStr?)

/-- A natural number in JSON. -/
def jNat (j : Json) : Except String Nat := j.getNat?

/-- The `i`-th element of an argument array. -/
def arg (a : Array Json) (i : Nat) : Except String Json :=
  match a[i]? with
  | some j => .ok j
  | none => .error s!"missing argument {i}"

/-- Parse one line. -/
def parseEvent (line : Nat) (s : String) : Except String Event := do
  let j ← Json.parse s
  let args ← match ← j.getObjVal? "args" with
    | .arr a => pure a
    | _ => throw "args is not an array"
  return {
    line
    session := ← (← j.getObjVal? "s").getNat?
    party := ← (← j.getObjVal? "p").getStr?
    op := ← (← j.getObjVal? "op").getNat?
    f := ← (← j.getObjVal? "f").getStr?
    args
    ret := ← j.getObjVal? "ret" }

/-- The name of a state, from the `States` variant name. -/
def parseLabel (s : String) : Except String Label :=
  match s with
  | "KeysUnsampled" => .ok .keysUnsampled
  | "KeysSampled" => .ok .keysSampled
  | "HeaderSent" => .ok .headerSent
  | "Ct1Received" => .ok .ct1Received
  | "EkSentCt1Received" => .ok .ekSentCt1Received
  | "NoHeaderReceived" => .ok .noHeaderReceived
  | "HeaderReceived" => .ok .headerReceived
  | "Ct1Sampled" => .ok .ct1Sampled
  | "EkReceivedCt1Sampled" => .ok .ekReceivedCt1Sampled
  | "Ct1Acknowledged" => .ok .ct1Acknowledged
  | "Ct2Sampled" => .ok .ct2Sampled
  | _ => .error s!"unknown state {s}"

/-- A chunk `{index, data}`. -/
def parseChunk (j : Json) : Except String MChunk := do
  return ⟨← jNat (← j.getObjVal? "index"), ← jBytes (← j.getObjVal? "data")⟩

/-- A message `{epoch, kind, ...}`. As in `absMsg` (`Relation.lean`), `Ct1Ack(false)` is the
`None` payload. -/
def parseMsg (j : Json) : Except String (Msg MChunk) := do
  let epoch ← jNat (← j.getObjVal? "epoch")
  let payload ← match ← (← j.getObjVal? "kind").getStr? with
    | "none" => pure Payload.none
    | "ct1ackfalse" => pure Payload.none
    | "ct1ack" => pure Payload.ct1Ack
    | "hdr" => pure (Payload.hdr (← parseChunk j))
    | "ek" => pure (Payload.ek (← parseChunk j))
    | "ekct1ack" => pure (Payload.ekCt1Ack (← parseChunk j))
    | "ct1" => pure (Payload.ct1 (← parseChunk j))
    | "ct2" => pure (Payload.ct2 (← parseChunk j))
    | k => throw s!"unknown message kind {k}"
  return ⟨epoch, payload⟩

/-- A key: `null` or `{epoch, secret}`. -/
def parseKey (j : Json) : Except String (Option Key) := do
  if j.isNull then return none
  return some (← jNat (← j.getObjVal? "epoch"), ← jBytes (← j.getObjVal? "secret"))

/-- The observation of a step event, with the randomness `r` of its operation. -/
def parseObs (e : Event) (r : Bytes) : Except String (Obs MChunk) := do
  match e.f with
  | "States::send" =>
    if (e.ret.getObjVal? "err").isOk then throw "States::send returned an error"
    return .send r (← parseMsg (← e.ret.getObjVal? "msg")) (← parseKey (← e.ret.getObjVal? "key"))
      (← parseLabel (← (← e.ret.getObjVal? "state").getStr?)) (← jNat (← e.ret.getObjVal? "epoch"))
  | "States::recv" =>
    let m ← parseMsg (← arg e.args 2)
    if (e.ret.getObjVal? "err").isOk then return .recvErr m
    return .recvOk m (← parseKey (← e.ret.getObjVal? "key"))
      (← parseLabel (← (← e.ret.getObjVal? "state").getStr?)) (← jNat (← e.ret.getObjVal? "epoch"))
  | f => throw s!"{f} is not a step event"

/-- Whether an event ends an operation. -/
def Event.isStep (e : Event) : Bool := e.f == "States::send" || e.f == "States::recv"

end Monitor

import Monitor.Check

/-! # `spqr-monitor`: check event logs against `BraidModel`

Usage: `spqr-monitor [--max-ops N] <trace.jsonl>...`

For each file and each session in it, the program checks both parties (`Monitor.checkParty`)
and the session-level properties (`Monitor.sessionChecks`), and prints one line per party and
per check. With `--max-ops N` it checks only the events up to the `N`-th operation of each
session, and says so: that is partial coverage of the file. The exit code is 0 when every
check of every file passes, 3 when every check passes on a prefix (`--max-ops`), and 1
otherwise. -/

open Lean Std Monitor

/-- Read a file into events. -/
def readEvents (path : String) : IO (Array Event) := do
  let text ← IO.FS.readFile path
  let mut evs : Array Event := #[]
  let mut n := 0
  for line in text.splitOn "\n" do
    n := n + 1
    if line.isEmpty then continue
    match parseEvent n line with
    | .ok e => evs := evs.push e
    | .error err => throw (IO.userError s!"{path}:{n}: {err}")
  return evs

/-- Print the result of a session check; return whether it passed. -/
def report (s : Nat) (what : String) (r : Except String Unit) : IO Bool := do
  match r with
  | .ok () => IO.println s!"  session {s} {what}: ok"; return true
  | .error err => IO.println s!"  session {s} {what}: FAILED: {err}"; return false

/-- Check one file; return whether every check passed. -/
def checkFile (path : String) (maxOps : Option Nat) : IO Bool := do
  let evs ← readEvents path
  let sessions := (evs.map (·.session)).toList.eraseDups
  let mut ok := true
  IO.println s!"{path}: {evs.size} events, {sessions.length} session(s)"
  if sessions.isEmpty then
    IO.println "  REJECTED: no events"
    return false
  for s in sessions do
    let mut sevs := evs.filter (·.session == s)
    if let some n := maxOps then
      let cut := sevs.filter (·.op ≤ n)
      if cut.size < sevs.size then
        IO.println s!"  session {s}: PARTIAL, only operations 0..{n} of this session are checked"
        -- keep whole operations only
        sevs := cut
    -- only parties A and B; operations numbered 0, 1, 2, ... in file order
    let mut last := 0
    for e in sevs do
      if e.party != "A" && e.party != "B" then
        ok := false
        IO.println s!"  session {s}: REJECTED: line {e.line}: unknown party {e.party}"
        break
      if e.op != last && e.op != last + 1 then
        ok := false
        IO.println s!"  session {s}: REJECTED: line {e.line}: operation {e.op} after \
          operation {last}"
        break
      last := e.op
    -- every operation other than the initializations (operation 0) belongs to one party
    let mut owner : HashMap Nat String := {}
    for e in sevs do
      if e.op == 0 then continue
      match owner[e.op]? with
      | some p =>
        if p != e.party then
          ok := false
          IO.println s!"  session {s}: REJECTED: line {e.line}: operation {e.op} has events of \
            parties {p} and {e.party}"
          break
      | none => owner := owner.insert e.op e.party
    let mut keys : HashMap String (List BraidModel.Key) := {}
    for p in ["A", "B"] do
      let pevs := sevs.filter (·.party == p)
      if pevs.isEmpty then
        ok := false
        IO.println s!"  session {s} party {p}: REJECTED: no events for this party"
        continue
      let r := checkParty p pevs
      keys := keys.insert p r.keys
      match r.verdict with
      | .ok () =>
        IO.println s!"  session {s} party {p}: ACCEPTED ({r.operations} operations, \
          {r.keys.length} keys)"
      | .error err =>
        ok := false
        IO.println s!"  session {s} party {p}: REJECTED: {err}"
    let c := sessionChecks sevs (keys.getD "A" []) (keys.getD "B" [])
    ok := (← report s "delivery" c.delivery) && ok
    ok := (← report s "key agreement" c.keys) && ok
    if !c.incompleteEpochs.isEmpty then
      IO.println s!"  session {s} key agreement: incomplete epochs (one party only) \
        {c.incompleteEpochs}"
    ok := (← report s "randomness reuse (outside BraidModel)" c.randomness) && ok
  return ok

def main (args : List String) : IO UInt32 := do
  let rec parse : List String → Option Nat → List String → Except String (Option Nat × List String)
    | [], m, fs => .ok (m, fs.reverse)
    | "--max-ops" :: n :: rest, _, fs => match n.toNat? with
      | some k => parse rest (some k) fs
      | none => .error s!"bad --max-ops {n}"
    | f :: rest, m, fs => parse rest m (f :: fs)
  match parse args none [] with
  | .error err => IO.eprintln err; return 2
  | .ok (_, []) => IO.eprintln "usage: spqr-monitor [--max-ops N] <trace.jsonl>..."; return 2
  | .ok (maxOps, files) =>
    let mut ok := true
    for f in files do
      ok := (← checkFile f maxOps) && ok
    if !ok then
      IO.println "SOME CHECKS FAILED"; return 1
    if maxOps.isSome then
      IO.println "ALL CHECKS PASSED ON THE CHECKED PREFIX (partial coverage, exit code 3)"
      return 3
    IO.println "ALL CHECKS PASSED"
    return 0

pub struct GenericMessagingScka<SCKA: Scka> {
    scka: SCKA,
    send_outputs: BTreeMap<Epoch, Secret>,
    recv_outputs: BTreeMap<Epoch, Secret>,
    last_emitted_epoch: Epoch,
}

#[cfg(feature = "monitor")]
impl<SCKA: Scka> GenericMessagingScka<SCKA> {
    /// The state of the underlying SCKA.
    pub fn scka(&self) -> &SCKA {
        &self.scka
    }
}

impl<SCKA: Scka + SckaInitializer + Clone> MessagingScka for GenericMessagingScka<SCKA> {
    type CkaOutput = Secret;

    type Message = SCKA::Message;

    fn init_a<R: CryptoRng>(rng: &mut R) -> Result<Self, Error>
    where
        Self: Sized,
    {
        Ok(Self {
            scka: SCKA::init_a(rng)?,
            send_outputs: BTreeMap::new(),
            recv_outputs: BTreeMap::new(),
            last_emitted_epoch: 0,
        })
    }

    fn init_b<R: CryptoRng>(rng: &mut R) -> Result<Self, Error>
    where
        Self: Sized,
    {
        Ok(Self {
            scka: SCKA::init_b(rng)?,
            send_outputs: BTreeMap::new(),
            recv_outputs: BTreeMap::new(),
            last_emitted_epoch: 0,
        })
    }

    fn messaging_scka_send<R: CryptoRng>(
        &mut self,
        rng: &mut R,
    ) -> Result<(Option<(Epoch, Self::CkaOutput)>, Self::Message), crate::Error>
    where
        Self::CkaOutput: Sized,
    {
        let (so, msg, state) = self.scka.clone().scka_send(rng)?;
        self.scka = state;

        // self.last_emitted_epoch = so.sending_epoch;
        let earliest_send_output = if let Some((ep, _)) = self.send_outputs.first_key_value() {
            *ep
        } else {
            0
        };

        if let Some((ep, k)) = so.output_key {
            self.send_outputs.insert(ep, k);
        }

        let mut take_key = false;
        if let Some(entry) = self.recv_outputs.first_entry() {
            let ep = *entry.key();
            if ep <= so.sending_epoch
                && (earliest_send_output == 0 || ep < earliest_send_output)
                && (self.last_emitted_epoch == 0 || ep == self.last_emitted_epoch + 1)
            {
                take_key = true;
            }
        };
        let output_key = if take_key {
            let entry = self.recv_outputs.first_entry().unwrap();
            let ep = *entry.key();
            self.last_emitted_epoch = ep;
            // info!("messaging scka send outputs: {:?}", entry);
            Some((ep, entry.remove()))
        } else {
            None
        };

        Ok((output_key, msg))
    }

    fn messaging_scka_recv(
        &mut self,
        msg: &Self::Message,
        _rng: &mut OsRng,
    ) -> Result<Option<(Epoch, Self::CkaOutput)>, Error>
    where
        Self::CkaOutput: Sized,
    {
        let (ro, state) = self.scka.clone().scka_recv(msg)?;
        self.scka = state;

        // self.last_emitted_epoch = ro.receiving_epoch;
        let earliest_recv_output = if let Some((ep, _)) = self.recv_outputs.first_key_value() {
            *ep
        } else {
            0
        };

        if let Some((ep, k)) = ro.output_key {
            self.recv_outputs.insert(ep, k);
        }

        let mut take_key = false;
        if let Some(entry) = self.send_outputs.first_entry() {
            let ep = *entry.key();
            if ep <= ro.receiving_epoch
                && (earliest_recv_output == 0 || ep < earliest_recv_output)
                && (self.last_emitted_epoch == 0 || ep == self.last_emitted_epoch + 1)
            {
                take_key = true;
            }
        };

        let output_key = if take_key {
            let entry = self.send_outputs.first_entry().unwrap();
            let ep = *entry.key();
            self.last_emitted_epoch = ep;
            // info!("messaging scka recv outputs: {:?}", entry);
            Some((ep, entry.remove()))
        } else {
            None
        };

        Ok(output_key)
    }
}

pub struct HealingStats {
    pub mean: f64,
    pub stddev: f64,
    pub deciles: [usize; 11],
}

pub fn stats_from_histogram(hist: &Vec<HealingHistogramEntry>) -> [HealingStats; 2] {
    // first pass: compute aggregates: count, sum(num_msgs), sum(num_msgs^2)
    let mut count = [0usize; 2];
    let mut sum = [0usize; 2];
    let mut sum_squares = [0usize; 2];
    let mut min = [usize::MAX; 2];
    let mut max = [0usize; 2];
    let mut deciles = [[0usize; 11]; 2];

    let mut mean = [0f64; 2];
    let mut mean_square = [0f64; 2];
    let mut var = [0f64; 2];
    let mut stddev = [0f64; 2];

    for entry in hist {
        count[0] += entry.tot_by_a;
        count[1] += entry.tot_by_b;
        sum[0] += entry.num_msgs * entry.tot_by_a;
        sum[1] += entry.num_msgs * entry.tot_by_b;
        sum_squares[0] += entry.num_msgs * entry.num_msgs * entry.tot_by_a;
        sum_squares[1] += entry.num_msgs * entry.num_msgs * entry.tot_by_b;
        if entry.tot_by_a > 0 {
            min[0] = std::cmp::min(min[0], entry.num_msgs);
            max[0] = std::cmp::max(max[0], entry.num_msgs);
        }
        if entry.tot_by_b > 0 {
            min[1] = std::cmp::min(min[1], entry.num_msgs);
            max[1] = std::cmp::max(max[1], entry.num_msgs);
        }
    }
    for i in 0..2 {
        mean[i] = (sum[i] as f64) / (count[i] as f64);
        mean_square[i] = (sum_squares[i] as f64) / (count[i] as f64);
        var[i] = mean_square[i] - mean[i] * mean[i];
        stddev[i] = var[i].sqrt();
    }

    for i in 0..2 {
        deciles[i][0] = min[i];
        deciles[i][10] = max[i];
        let mut cummulative_count = 0usize;
        let mut ctr = 1usize;
        for entry in hist {
            cummulative_count += if i == 0 {
                entry.tot_by_a
            } else {
                entry.tot_by_b
            };
            let decile_target = (count[i] * ctr) / 10;
            if cummulative_count > decile_target {
                deciles[i][ctr] = entry.num_msgs;
                ctr += 1;
            }
            if ctr == 10 {
                break;
            }
        }
    }

    [
        HealingStats {
            mean: mean[0],
            stddev: stddev[0],
            deciles: deciles[0],
        },
        HealingStats {
            mean: mean[1],
            stddev: stddev[1],
            deciles: deciles[1],
        },
    ]
}

pub fn print_histogram(hist: &Vec<HealingHistogramEntry>) {
    println!(
        "num exposed,msgs exposed by a comp,msgs exposed by b comp,freq a exposed by full,freq b exposed by full,tot exposed by full"
    );
    for entry in hist {
        println!("{},{},{}", entry.num_msgs, entry.tot_by_a, entry.tot_by_b,);
    }
}

pub fn random_healing_test<CKA, R: CryptoRng>(ratio: f64, rng: &mut R) -> Result<(), Error>
where
    CKA: MessagingScka + MessagingCkaVulnerability,
    <CKA as MessagingScka>::CkaOutput: PartialEq + Debug,
{
    let base_send_prob = 0.5;
    let p_a = base_send_prob * ratio;
    let p_b = base_send_prob * (1.0 - ratio);
    let mut mp = BasicMessagingBehavior::new(p_a, p_b, 0.9);
    let hist =
        controlled_messaging_healing_test::<CKA, BasicMessagingBehavior, R>(&mut mp, 10000, rng)?;
    print_histogram(&hist);
    print_healing_stats(&stats_from_histogram(&hist)[0]);
    Ok(())
}

pub fn random_balanced<CKA, R>(rng: &mut R) -> Result<(), Error>
where
    CKA: MessagingScka + MessagingCkaVulnerability,
    <CKA as MessagingScka>::CkaOutput: PartialEq + Debug,
    R: CryptoRng,
{
    let mut a_tot = 0;
    let mut b_tot = 0;
    let mut orchestrator = OrchestratorBase::<CKA>::new(rng)?;
    for _i in 0..10000 {
        let rnd: u32 = rng.next_u32();
        let use_alex = rnd & 0x1 != 0;
        let do_receive = rnd & 0x6 != 0;
        let do_send = rnd & 0x8 != 0;
        if do_receive {
            // orchestrator.receive_in_order(use_alex)?;
            orchestrator.receive_all(use_alex)?;
        }
        if do_send {
            orchestrator.send(use_alex, rng)?;
            if use_alex {
                a_tot += 1;
            } else {
                b_tot += 1;
            }
        }
        orchestrator.key_history_is_consistent();
    }
    orchestrator.print_key_history_lengths();
    orchestrator.print_msg_queue_lengths();
    println!("Alex sent {a_tot}  Blake sent {b_tot}");
    Ok(())
}

pub fn random_balanced_out_of_order<CKA, R>(rng: &mut R) -> Result<(), Error>
where
    CKA: MessagingScka + MessagingCkaVulnerability,
    <CKA as MessagingScka>::CkaOutput: PartialEq + Debug,
    R: CryptoRng,
{
    let mut orchestrator = OrchestratorBase::<CKA>::new(rng)?;
    for _i in 0..10000 {
        let rnd = rng.next_u32();
        let use_alex = rnd & 0x1 == 0;
        let do_send = rnd & 0x2 != 0;
        let do_ooo = rnd & 0x4 == 0;
        let rcv_all = rnd & 0xF8 == 0;
        if do_ooo {
            if orchestrator.qlen(use_alex) > 0 {
                orchestrator.receive_at(
                    use_alex,
                    ((rnd >> 8) as usize) % orchestrator.qlen(use_alex),
                )?;
            }
        } else if rcv_all {
            orchestrator.receive_all(use_alex)?;
        } else {
            orchestrator.receive_in_order(use_alex)?;
        }
        if do_send {
            orchestrator.send(use_alex, rng)?;
        }
        orchestrator.key_history_is_consistent();
    }
    orchestrator.print_key_history_lengths();
    orchestrator.print_msg_queue_lengths();
    Ok(())
}

pub fn chaos<CKA, R>(num_ticks: usize, rng: &mut R) -> Result<(), Error>
where
    CKA: MessagingScka + MessagingCkaVulnerability,
    <CKA as MessagingScka>::CkaOutput: PartialEq + Debug,
    R: CryptoRng,
{
    let ooo_prob = 0.7;
    let send_limit = 10;
    let drop_message_prob = 0.1;
    let mut orchestrator = OrchestratorBase::<CKA>::new(rng)?;
    for i in 0..num_ticks {
        let use_alex = if i % 100 < 50 {
            rng.random_bool(0.25)
        } else {
            rng.random_bool(0.75)
        };

        // receive out of order
        if rng.random_bool(ooo_prob) {
            let qlen = orchestrator.qlen(use_alex);
            if qlen > 0 {
                let _received =
                    orchestrator.receive_at(use_alex, rng.next_u32() as usize % qlen)?;
            }
        }
        let num_to_send = rng.next_u32() % send_limit;
        for _ in 0..num_to_send {
            orchestrator.send(use_alex, rng)?;
            if rng.random_bool(drop_message_prob) {
                orchestrator.drop_message_at(use_alex, 0);
            }
        }

        // don't let queue get too big
        loop {
            let qlen = orchestrator.qlen(use_alex);
            if qlen > 20 {
                let _received =
                    orchestrator.receive_at(use_alex, rng.next_u32() as usize % qlen)?;
            } else {
                break;
            }
        }

        orchestrator.key_history_is_consistent();
    }
    orchestrator.print_key_history_lengths();
    orchestrator.print_msg_queue_lengths();
    Ok(())
}

//! Event log for the runtime monitor (feature `monitor`).
//!
//! Each event is one JSON line `{"s":..,"p":..,"op":..,"f":..,"args":..,"ret":..}`: the session
//! number, the party (`A` or `B`), the operation number, the function name, its arguments and
//! its result, with bytes as lowercase hex. Operation 0 is a party's initialization; the test
//! harness starts operation `n + 1` before each `States::send` or `States::recv` call
//! (`begin_op`), and the call's step event ends it. A test that tampers with a message in

#[derive(Default)]
struct Sink {
    session: u64,
    op: u64,
    party: Option<char>,
    out: Option<std::io::BufWriter<std::fs::File>>,
    disabled: bool,
}

impl Sink {
    #[cfg(test)]
    fn flush(&mut self) {
        if let Some(out) = self.out.as_mut() {
            out.flush().expect("monitor: cannot flush the event log");
        }
    }

    fn open(&mut self) -> bool {
        if self.out.is_some() {
            return true;
        }
        if self.disabled {
            return false;
        }
        let Some(dir) = std::env::var_os("SPQR_MONITOR_DIR") else {
            self.disabled = true;
            return false;
        };
        let name = std::thread::current()
            .name()
            .unwrap_or("unnamed")
            .replace("::", ".");
        std::fs::create_dir_all(&dir).expect("monitor: cannot create SPQR_MONITOR_DIR");
        let path = std::path::Path::new(&dir).join(format!("{name}.jsonl"));
        let file = std::fs::File::create(&path).expect("monitor: cannot create the event log");
        self.out = Some(std::io::BufWriter::new(file));
        true
    }
}

impl Drop for Sink {
    fn drop(&mut self) {
        if let Some(out) = self.out.as_mut() {
            let _ = out.flush();
        }
    }
}

/// Write an event in operation `op + offset`, where `op` is the current operation.
fn emit_at(offset: u64, f: &str, args: &str, ret: &str) {
    SINK.with(|s| {
        let mut s = s.borrow_mut();
        let Some(p) = s.party else { return };
        if !s.open() {
            return;
        }
        let (session, op) = (s.session, s.op + offset);
        let out = s.out.as_mut().expect("opened above");
        writeln!(
            out,
            "{{\"s\":{session},\"p\":\"{p}\",\"op\":{op},\"f\":\"{f}\",\"args\":{args},\"ret\":{ret}}}"

    });

fn hex(bytes: &[u8]) -> String {
    let mut s = String::with_capacity(2 * bytes.len() + 2);
    s.push('"');
    for b in bytes {
        write!(s, "{b:02x}").expect("writing to a String");
    }
    s.push('"');
    s
}

/// Pt is an internal cartesian point for a function in the GF16
/// space.  It's mostly used to allow for lookup and ordering by
/// the X value in a BTreeSet.
struct Pt {
    x: GF16,
    y: GF16,
}

impl Pt {
    fn serialize(&self) -> [u8; 4] {
        let mut out = [0u8; 4];
        out[..2].clone_from_slice(&self.x.value.to_be_bytes()[..]);
        out[2..].clone_from_slice(&self.y.value.to_be_bytes()[..]);
        out
    }
    fn deserialize(s: [u8; 4]) -> Self {
        Self {
            x: GF16::new(u16::from_be_bytes(s[..2].try_into().unwrap())),
            y: GF16::new(u16::from_be_bytes(s[2..].try_into().unwrap())),
        }
    }
}

impl Ord for Pt {
    fn cmp(&self, other: &Self) -> Ordering {
        self.partial_cmp(other).unwrap()
    }
}

#[derive(Clone, PartialEq)]
#[hax_lib::attributes]
pub(crate) struct Poly {
    // For Protocol V1 we interpolate at most 36 values, which produces a
    // degree 35 polynomial (with 36 coefficients). In an intermediate calculation
    // during Lagrange interpolation, we need to compute a polynomial one degree
    // higher, thus we get the following constraint:
    //
    // coefficients.len() <= MAX_INTERMEDIATE_POLYNOMIAL_DEGREE_V1 + 1
    #[hax_lib::refine(coefficients.len() <= MAX_INTERMEDIATE_POLYNOMIAL_DEGREE_V1 + 1)]
    pub coefficients: Vec<GF16>,
}

#[hax_lib::attributes]
impl Poly {
    //
    // capacity <= MAX_INTERMEDIATE_POLYNOMIAL_DEGREE_V1 + 1
    fn zero(capacity: usize) -> Self {
        Self {
            coefficients: Vec::with_capacity(capacity),
        }
    }

    /// Given a set of points with unique X values, return a Poly that
    /// computes f(pts[i].x) == pts[i].y for all points.
    ///
    /// This takes O(N^2) work and O(N) space, carefully allocated up
    /// front to avoid overhead.
    #[hax_lib::requires(pts.len() <= MAX_INTERMEDIATE_POLYNOMIAL_DEGREE_V1)]
    fn lagrange_interpolate(pts: &[Pt]) -> Self {
        let mut out = Self::zero(pts.len());
        if pts.is_empty() {
            return out;
        }

        let template = Self::lagrange_interpolate_prepare(pts);
        let mut working = template.clone();

        // Unroll the first loop to skip some unnecessary work.
        working.lagrange_interpolate_complete(pts, 0);
        // Note that `working` is `x * <the polynomial we need>`.
        out.coefficients
            .extend_from_slice(&working.coefficients.as_slice()[1..]);

        let _w_l = working.coefficients.len();
        for i in 1..pts.len() {
            hax_lib::loop_invariant!(
                |_: usize| out.coefficients.len() == _w_l - 1 && working.coefficients.len() == _w_l
            );
            working.coefficients.copy_from_slice(&template.coefficients);
            working.lagrange_interpolate_complete(pts, i);
            // We can't use `add_assign` because `working` is `x * <the polynomial we need>`.
            // Removing the lowest coefficient would cost a memmove.
            // So we just skip it in this loop to effectively "divide by x".
            for j in 0..out.coefficients.len() {
                hax_lib::loop_invariant!(|_: usize| out.coefficients.len() == _w_l - 1);
                out.coefficients[j] += working.coefficients[j + 1];
            }
        }
        out
    }

    /// Computes `PRODUCT(x - pi.x)` as part of Lagrange interpolation.
    ///
    /// This takes O(N^2) work: for each of N points, we are multiplying every coefficient of a
    /// 1..N-degree polynomial.
    #[hax_lib::requires(pts.len() <= MAX_INTERMEDIATE_POLYNOMIAL_DEGREE_V1)]
    fn lagrange_interpolate_prepare(pts: &[Pt]) -> Self {
        // We're going to fill in the coefficients from largest to smallest, so we start by putting
        // a 1 in the *highest* field rather than the lowest. This lets us avoid sliding
        // coefficients as we go, but it also means we have to track the offset of which
        // coefficients have been initialized manually.
        let mut p = Self {
            coefficients: vec![GF16::ZERO; pts.len() + 1],
        };
        let offset = pts.len();
        p.coefficients[offset] = GF16::ONE;

        #[allow(clippy::needless_range_loop)]
        for i in 0..offset {
            hax_lib::loop_invariant!(|_: usize| p.coefficients.len() == offset + 1);
            let pi = pts[i];
            p.mult_xdiff_assign_trailing(offset - i, pi.x);
        }
        #[cfg(not(hax))]
        debug_assert_eq!(p.coefficients[pts.len()], GF16::ONE);
        p
    }

    /// self[start..] *= (x - difference)
    ///
    /// Interprets the trailing N coefficients as a smaller polynomial, and multiplies *that*
    /// polynomial by `(x - difference)`, with the "carry" propagating into self[start-1]. This
    /// works because (x-d)(poly) = x*poly - d*poly.
    ///
    /// This allows us to build up a polynomial from its *largest* coefficient, and thus avoid
    /// sliding coefficients in the vector as we go.
    #[hax_lib::requires(0 < start && start <= self.coefficients.len())]
    fn mult_xdiff_assign_trailing(&mut self, start: usize, difference: GF16) {
        let l = self.coefficients.len();
        for i in start..l {
            hax_lib::loop_invariant!(|_: usize| self.coefficients.len() == l);
            let delta = self.coefficients[i] * difference;
            self.coefficients[i - 1] -= delta;
        }
    }

    /// `pts`.
    ///

    /// the remaining polynomial produces `pi.y` at `pi.x`.

    ///

        // Compute the scaling factor.
        let mut denominator = GF16::ONE;
        for pj in pts {
            if pi.x == pj.x {
                continue;
            }
            denominator *= pi.x - pj.x;
        }

        // This avoids having to reload the same value from memory twice.

        let _init_l = self.coefficients.len();

        }

    pub fn send<R: CryptoRng>(&mut self, is_alex: bool, rng: &mut R) -> Result<bool, Error> {
        #[cfg(feature = "monitor")]
        crate::monitor::set_party(is_alex);
        let mut emitted_key = false;
        let mut emitted_ep: Option<Epoch> = None;
        {
            {
                let (agent, _incoming_msg_queue, outgoing_msg_queue, key_history) =
                    self.data_for_agent(is_alex);
                let (out, msg) = agent.messaging_scka_send(rng)?;
                if let Some((_ep, key)) = out {
                    key_history.push(key);
                    emitted_ep = Some(agent.last_emitted_epoch());
                    emitted_key = true;
                }
                outgoing_msg_queue.push_back(msg);
            }
            if is_alex {
                self.a_sent += 1;
            } else {
                self.b_sent += 1;
            }
        }

        if let Some(ep) = emitted_ep {
            if is_alex {
                self.last_emitted_epoch_a = ep;
            } else {
                self.last_emitted_epoch_b = ep;
            }
        }
        Ok(emitted_key)
    }

    pub fn receive_all(&mut self, is_alex: bool) -> Result<bool, Error> {
        #[cfg(feature = "monitor")]
        crate::monitor::set_party(is_alex);
        let mut emitted_ep: Option<Epoch> = None;
        let mut emitted_key = false;
        let mut num_received = 0usize;
        let (agent, incoming_message_queue, _omq, key_history) = self.data_for_agent(is_alex);
        while !incoming_message_queue.is_empty() {
            let maybe_msg = incoming_message_queue.pop_front();
            if let Some(msg) = maybe_msg {
                let mut rng = OsRng;
                let out = agent.messaging_scka_recv(&msg, &mut rng)?;
                if let Some((_ep, key)) = out {
                    key_history.push(key);
                    emitted_key = true;
                    emitted_ep = Some(agent.last_emitted_epoch());
                }
                num_received += 1;
            }
        }
        if is_alex {
            self.a_rcvd += num_received;
        } else {
            self.b_rcvd += num_received;
        }
        if let Some(ep) = emitted_ep {
            if is_alex {
                self.last_emitted_epoch_a = ep;
            } else {
                self.last_emitted_epoch_b = ep;
            }
        }
        Ok(emitted_key)

    pub fn last_vulnerable_epoch_a(&self) -> Epoch {
        *self.alex.vulnerable_epochs().iter().max().unwrap_or(&0u64)
    {
        let (ro, state) = self.scka.clone().scka_recv(msg)?;
        self.scka = state;

        // self.last_emitted_epoch = ro.receiving_epoch;
        let earliest_recv_output = if let Some((ep, _)) = self.recv_outputs.first_key_value() {
            *ep
        } else {
            0
        };

        if let Some((ep, k)) = ro.output_key {
            self.recv_outputs.insert(ep, k);
        }

        let mut take_key = false;
        if let Some(entry) = self.send_outputs.first_entry() {
            let ep = *entry.key();
            if ep <= ro.receiving_epoch
                && (earliest_recv_output == 0 || ep < earliest_recv_output)
                && (self.last_emitted_epoch == 0 || ep == self.last_emitted_epoch + 1)
            {
                take_key = true;
            }
        };

        let output_key = if take_key {
            let entry = self.send_outputs.first_entry().unwrap();
            let ep = *entry.key();
            self.last_emitted_epoch = ep;
            // info!("messaging scka recv outputs: {:?}", entry);
            Some((ep, entry.remove()))
        } else {
            None
        };

        Ok(output_key)
    }
